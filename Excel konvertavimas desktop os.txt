import os
import win32com.client

# Pagrindinis katalogas, kuriame ieškosime .xls failų
main_directory = r"C:\Users\sigitaabasoviene\OneDrive - Corpus A, UAB\Desktop\Aktai"

# Sukuriame Excel COM objektą vieną kartą
excel = win32com.client.Dispatch("Excel.Application")
excel.Visible = False
excel.DisplayAlerts = False
excel.ScreenUpdating = False

# Funkcija konvertuoja .xls į .xlsx
def convert_xls_to_xlsx(file_path):
    new_file_path = os.path.splitext(file_path)[0] + ".xlsx"
    if os.path.exists(new_file_path):  # Jei naujas failas jau egzistuoja, praleiskime
        print(f"Failas jau egzistuoja: {new_file_path}")
        return
    try:
        wb = excel.Workbooks.Open(file_path)  # Atidarykite .xls failą
        wb.SaveAs(new_file_path, FileFormat=51)  # 51 - tai xlsx formatas
        wb.Close()
        print(f"Konvertuotas failas: {file_path} -> {new_file_path}")
    except Exception as e:
        print(f"Klaida konvertuojant failą {file_path}: {e}")

# Pereiname per visus aplankus ir failus
for root, dirs, files in os.walk(main_directory):
    for file in files:
        if file.endswith(".xls"):  # Tikriname, ar failas yra .xls formato
            file_path = os.path.join(root, file)
            convert_xls_to_xlsx(file_path)

# Uždarykite Excel, kai baigsite
excel.Quit()
print("Konvertavimas baigtas.")
